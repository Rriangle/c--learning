什麼是物件導向的三大特性？

物件導向程式設計有三大核心特性：封裝（Encapsulation）、繼承（Inheritance）、多型（Polymorphism）。這三個特性讓我們可以建立更模組化、可重用和易於維護的程式碼。

三大特性：
1. 封裝：隱藏內部實作細節，只暴露必要的介面
2. 繼承：子類別可以繼承父類別的屬性和方法
3. 多型：同一個介面可以有不同的實作方式

封裝（Encapsulation）

封裝就像是把重要的東西放在保險箱裡，只有擁有鑰匙的人才能存取。在程式設計中，我們使用存取修飾詞來控制類別成員的可見性。

using System; // 示範封裝的銀行帳戶類別 public class BankAccount { // 私有欄位 - 外部無法直接存取 private decimal balance; private string accountNumber; private string ownerName; private bool isActive; // 公開屬性 - 提供受控的存取方式 public decimal Balance { get { return balance; } // 唯讀屬性 private set { balance = value; } // 只有內部可以設定 } public string AccountNumber { get { return accountNumber; } // 唯讀屬性 } public string OwnerName { get { return ownerName; } set { if (!string.IsNullOrEmpty(value)) ownerName = value; } } public bool IsActive { get { return isActive; } private set { isActive = value; } // 只有內部可以設定 } // 建構子 public BankAccount(string owner, decimal initialBalance) { ownerName = owner; balance = initialBalance; accountNumber = GenerateAccountNumber(); isActive = true; Console.WriteLine($"🏦 建立帳戶：{accountNumber}（持有人：{owner}）"); } // 私有方法 - 外部無法呼叫 private string GenerateAccountNumber() { Random random = new Random(); return $"ACC{random.Next(100000, 999999)}"; } // 公開方法 - 提供受控的操作介面 public bool Deposit(decimal amount) { if (!isActive) { Console.WriteLine("❌ 帳戶已被停用"); return false; } if (amount <= 0) { Console.WriteLine("❌ 存款金額必須大於零"); return false; } balance +=amount; Console.WriteLine($"✅ 存款 ${amount:N2}，新餘額：${balance:N2}"); return true; } public bool Withdraw(decimal amount) { if (!isActive) { Console.WriteLine("❌ 帳戶已被停用"); return false; } if (amount <=0) { Console.WriteLine("❌ 提款金額必須大於零"); return false; } if (amount> balance) { Console.WriteLine($"❌ 餘額不足，無法提款 ${amount:N2}"); return false; } balance -= amount; Console.WriteLine($"✅ 提款 ${amount:N2}，新餘額：${balance:N2}"); return true; } // 內部方法 - 只有同一個組件內可以呼叫 internal void CloseAccount() { isActive = false; Console.WriteLine($"🔒 關閉帳戶：{accountNumber}"); } public void ShowInfo() { Console.WriteLine($"📋 帳戶資訊："); Console.WriteLine($" 帳號：{accountNumber}"); Console.WriteLine($" 持有人：{ownerName}"); Console.WriteLine($" 餘額：${balance:N2}"); Console.WriteLine($" 狀態：{(isActive ? "啟用" : "停用")}"); } } // 主程式 class Program { static void Main() { Console.WriteLine("=== 封裝示範 ==="); // 建立銀行帳戶 BankAccount account = new BankAccount("小明", 10000); // 使用公開介面操作帳戶 account.ShowInfo(); Console.WriteLine(); // 存款和提款操作 account.Deposit(5000); account.Withdraw(2000); account.Withdraw(20000); // 餘額不足 Console.WriteLine(); // 嘗試直接存取私有成員（這會造成編譯錯誤） // account.balance = 1000000; // 錯誤！無法存取私有欄位 // account.accountNumber = "123"; // 錯誤！無法存取私有欄位 // 但可以透過公開屬性存取 Console.WriteLine($"目前餘額：${account.Balance}"); Console.WriteLine($"帳號：{account.AccountNumber}"); // 可以修改公開屬性 account.OwnerName = "小華"; Console.WriteLine($"新持有人：{account.OwnerName}"); // 但無法修改唯讀屬性 // account.Balance = 50000; // 錯誤！Balance 是唯讀的 Console.WriteLine(); // 關閉帳戶 account.CloseAccount(); account.Deposit(1000); // 無法存款，因為帳戶已關閉 Console.WriteLine(); account.ShowInfo(); } }

繼承（Inheritance）

繼承就像是「子承父業」，子類別可以繼承父類別的所有公開和受保護的成員，並且可以添加自己的新功能或覆寫父類別的方法。

using System; // 基底類別（父類別） public class Animal { // 受保護的欄位 - 子類別可以存取 protected string name; protected int age; protected string species; // 公開屬性 public string Name { get { return name; } set { name = value; } } public int Age { get { return age; } set { age = value; } } public string Species { get { return species; } set { species = value; } } // 建構子 public Animal(string name, int age, string species) { this.name = name; this.age = age; this.species = species; Console.WriteLine($"🐾 建立動物：{name}（{species}，{age}歲）"); } // 虛擬方法 - 子類別可以覆寫 public virtual void MakeSound() { Console.WriteLine($"{name} 發出聲音"); } // 一般方法 public void Sleep() { Console.WriteLine($"{name} 正在睡覺..."); } public void Eat() { Console.WriteLine($"{name} 正在吃東西..."); } // 虛擬方法 - 子類別可以覆寫 public virtual void ShowInfo() { Console.WriteLine($"📋 動物資訊："); Console.WriteLine($" 姓名：{name}"); Console.WriteLine($" 年齡：{age}歲"); Console.WriteLine($" 種類：{species}"); } } // 子類別 - 繼承 Animal public class Dog : Animal { // 子類別特有的欄位 private string breed; private bool isTrained; // 子類別特有的屬性 public string Breed { get { return breed; } set { breed = value; } } public bool IsTrained { get { return isTrained; } set { isTrained = value; } } // 子類別建構子 - 呼叫父類別建構子 public Dog(string name, int age, string breed, bool isTrained) : base(name, age, "狗") // 呼叫父類別建構子 { this.breed = breed; this.isTrained = isTrained; Console.WriteLine($"🐕 這是一隻 {breed} 犬"); } // 覆寫父類別的方法 public override void MakeSound() { Console.WriteLine($"{name} 汪汪叫！"); } // 覆寫父類別的方法 public override void ShowInfo() { base.ShowInfo(); // 呼叫父類別的方法 Console.WriteLine($" 品種：{breed}"); Console.WriteLine($" 訓練狀態：{(isTrained ? "已訓練" : "未訓練")}"); } // 子類別特有的方法 public void Fetch() { if (isTrained) { Console.WriteLine($"{name} 正在撿球..."); } else { Console.WriteLine($"{name} 還不會撿球，需要訓練"); } } public void Train() { if (!isTrained) { isTrained = true; Console.WriteLine($"{name} 訓練完成！"); } else { Console.WriteLine($"{name} 已經訓練過了"); } } } // 另一個子類別 public class Cat : Animal { private string color; private bool isIndoor; public string Color { get { return color; } set { color = value; } } public bool IsIndoor { get { return isIndoor; } set { isIndoor = value; } } public Cat(string name, int age, string color, bool isIndoor) : base(name, age, "貓") { this.color = color; this.isIndoor = isIndoor; Console.WriteLine($"🐱 這是一隻 {color} 色的貓"); } public override void MakeSound() { Console.WriteLine($"{name} 喵喵叫！"); } public override void ShowInfo() { base.ShowInfo(); Console.WriteLine($" 顏色：{color}"); Console.WriteLine($" 飼養方式：{(isIndoor ? "室內貓" : "戶外貓")}"); } public void Climb() { Console.WriteLine($"{name} 正在爬樹..."); } public void Purr() { Console.WriteLine($"{name} 發出呼嚕聲..."); } } // 主程式 class Program { static void Main() { Console.WriteLine("=== 繼承示範 ==="); // 建立動物物件 Animal animal = new Animal("小動物", 3, "未知"); animal.ShowInfo(); animal.MakeSound(); Console.WriteLine(); // 建立狗物件 Dog dog = new Dog("小白", 2, "拉布拉多", false); dog.ShowInfo(); dog.MakeSound(); dog.Fetch(); dog.Train(); dog.Fetch(); Console.WriteLine(); // 建立貓物件 Cat cat = new Cat("咪咪", 1, "橘色", true); cat.ShowInfo(); cat.MakeSound(); cat.Climb(); cat.Purr(); Console.WriteLine(); // 多型示範 - 使用父類別參考子類別物件 Console.WriteLine("=== 多型示範 ==="); Animal[] animals = { animal, dog, cat }; foreach (Animal a in animals) { Console.WriteLine($"\n{a.Name} 的行為："); a.MakeSound(); // 會呼叫各自子類別的實作 a.Sleep(); a.Eat(); } Console.WriteLine(); // 檢查物件類型 foreach (Animal a in animals) { if (a is Dog) { Dog d = (Dog)a; // 型別轉換 d.Fetch(); } else if (a is Cat) { Cat c = (Cat)a; // 型別轉換 c.Climb(); } } Console.WriteLine(); // 使用 as 關鍵字進行安全的型別轉換 foreach (Animal a in animals) { Dog d = a as Dog; if (d != null) { d.Fetch(); } Cat c = a as Cat; if (c != null) { c.Climb(); } } } }

} } public string Model { get { return model; } set { model = value; } } public int Year { get { return year; } set { year = value; } } public double MaxSpeed { get { return maxSpeed; } set { maxSpeed = value; } } public Vehicle(string brand, string model, int year, double maxSpeed) { this.brand = brand; this.model = model; this.year = year; this.maxSpeed = maxSpeed; Console.WriteLine($"🚗 建立交通工具：{brand} {model} {year}年"); } // 抽象方法 - 子類別必須實作 public abstract void Move(); // 虛擬方法 - 子類別可以覆寫 public virtual void Stop() { Console.WriteLine($"{brand} {model} 停止移動"); } public virtual void ShowInfo() { Console.WriteLine($"📋 交通工具資訊：");
