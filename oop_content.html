🔨 建構子與方法多載 ▼

什麼是建構子多載？

建構子多載讓我們可以用不同的方式建立物件，就像可以用不同的食譜做同一道菜一樣。每個建構子可以接受不同的參數，提供不同的初始化方式。

重要概念：
1. 建構子多載：同一個類別有多個建構子
2. 方法多載：同一個方法名稱有多個版本
3. 參數數量：不同數量的參數
4. 參數類型：不同類型的參數
5. this 關鍵字：呼叫同類別的其他建構子

建構子多載範例

讓我們看看如何為一個類別定義多個建構子，提供不同的初始化選項。

"電動車" : "汽油車")}"); } // 方法多載 - 格式化輸出 public string ShowInfo(string format) { switch (format.ToLower()) { case "short": return $"{Brand} {Model}"; case "medium": return $"{Brand} {Model} {Year}"; case "full": return $"{Brand} {Model} {Year} {Color} ${Price:N0}"; default: return $"{Brand} {Model} {Year}"; } } // 覆寫 ToString 方法 public override string ToString() { return $"{Brand} {Model} {Year}"; } } // 主程式 class Program { static void Main() { Console.WriteLine("=== 建構子與方法多載示範 ==="); Car car3 = new Car("Honda", "Civic", 2023, "藍色", 25000); // 完整建構子 car3.ShowInfo(); Console.WriteLine(); Car car4 = new Car("Tesla", "Model 3", 2024, "白色", 45000, true); // 電動車建構子 car4.ShowInfo(); Console.WriteLine(); // 2. 複製建構子 Console.WriteLine("2. 複製建構子："); Car car5 = new Car(car3); // 複製 car3 car5.Color = "紅色"; // 修改顏色 car5.ShowInfo(); Console.WriteLine(); // 3. 方法多載示範 Console.WriteLine("3. 方法多載示範："); Console.WriteLine("基本資訊："); } } Console.WriteLine($"💰 最貴的汽車：{mostExpensive.ShowInfo("full")}"); Console.WriteLine(); // 5. 啟動所有汽車 Console.WriteLine("5. 啟動所有汽車："); foreach (Car car in cars) { car.StartEngine(); } Console.WriteLine(); using System; using System.Collections.Generic; // 帳戶類型列舉 public enum AccountType { Basic, // 基本帳戶 Savings, // 儲蓄帳戶 Checking // 支票帳戶 } // 銀行帳戶類別 public class BankAccount { // 屬性 public string AccountNumber { get; private set; } // 帳號（唯讀） public string OwnerName { get; set; } // 帳戶持有人 public decimal Balance { get; private set; } // 餘額（唯讀） public AccountType Type { get; set; } // 帳戶類型 public DateTime CreatedDate { get; private set; } // 開戶日期 public bool IsActive { get; set; } // 是否啟用 // 靜態屬性 - 用於產生帳號 private static int accountCounter = 1000; // 預設建構子 public BankAccount() { AccountNumber = GenerateAccountNumber(); OwnerName = "未知"; Balance = 0.0m; Type = AccountType.Basic; CreatedDate = DateTime.Now; IsActive = true; Console.WriteLine($"🏦 建立基本帳戶：{AccountNumber}"); } // 基本建構子 - 只有持有人姓名 public BankAccount(string ownerName) { AccountNumber = GenerateAccountNumber(); OwnerName = ownerName; Balance = 0.0m; Type = AccountType.Basic; CreatedDate = DateTime.Now; IsActive = true; Console.WriteLine($"🏦 建立基本帳戶：{AccountNumber}（持有人：{ownerName}）"); } // 完整建構子 - 所有基本資訊 public BankAccount(string ownerName, decimal initialBalance, AccountType type) { AccountNumber = GenerateAccountNumber(); OwnerName = ownerName; Balance = initialBalance; Type = type; CreatedDate = DateTime.Now; IsActive = true; Console.WriteLine($"🏦 建立{GetTypeName(type)}：{AccountNumber}（持有人：{ownerName}，初始餘額：${initialBalance:N2}）"); } // 複製建構子 public BankAccount(BankAccount otherAccount) { AccountNumber = GenerateAccountNumber(); OwnerName = otherAccount.OwnerName; Balance = otherAccount.Balance; Type = otherAccount.Type; CreatedDate = DateTime.Now; IsActive = true; Console.WriteLine($"🏦 複製帳戶：{AccountNumber}（從 {otherAccount.AccountNumber} 複製）"); } // 私有方法 - 產生帳號 private string GenerateAccountNumber() { return $"ACC{++accountCounter:D6}"; } // 私有方法 - 取得帳戶類型名稱 private string GetTypeName(AccountType type) { return type switch { AccountType.Basic => "基本帳戶", AccountType.Savings => "儲蓄帳戶", AccountType.Checking => "支票帳戶", _ => "未知帳戶" }; } // 存款方法 - 基本版本 public bool Deposit(decimal amount) { if (amount <= 0) { Console.WriteLine($"❌ 存款金額必須大於零"); return false; } if (!IsActive) { Console.WriteLine($"❌ 帳戶 {AccountNumber} 已被停用"); return false; } Balance += amount; Console.WriteLine($"✅ {OwnerName} 存款 ${amount:N2}，新餘額：${Balance:N2}"); return true; } // 存款方法多載 - 包含備註 public bool Deposit(decimal amount, string note) { bool success = Deposit(amount); if (success) { Console.WriteLine($"📝 備註：{note}"); } return success; } // 提款方法 - 基本版本 public bool Withdraw(decimal amount) { if (amount <= 0) { Console.WriteLine($"❌ 提款金額必須大於零"); return false; } if (!IsActive) { Console.WriteLine($"❌ 帳戶 {AccountNumber} 已被停用"); return false; } if (amount > Balance) { Console.WriteLine($"❌ 餘額不足，無法提款 ${amount:N2}（餘額：${Balance:N2}）"); return false; } Balance -= amount; Console.WriteLine($"✅ {OwnerName} 提款 ${amount:N2}，新餘額：${Balance:N2}"); return true; } // 提款方法多載 - 包含備註 public bool Withdraw(decimal amount, string note) { bool success = Withdraw(amount); if (success) { Console.WriteLine($"📝 備註：{note}"); } return success; } // 轉帳方法 - 轉出 public bool Transfer(BankAccount targetAccount, decimal amount) { if (targetAccount == null) { Console.WriteLine($"❌ 目標帳戶不存在"); return false; } if (!targetAccount.IsActive) { Console.WriteLine($"❌ 目標帳戶 {targetAccount.AccountNumber} 已被停用"); return false; } if (Withdraw(amount)) { targetAccount.Deposit(amount); Console.WriteLine($"✅ 成功轉帳 ${amount:N2} 到 {targetAccount.OwnerName} 的帳戶"); return true; } return false; } // 轉帳方法多載 - 包含備註 public bool Transfer(BankAccount targetAccount, decimal amount, string note) { bool success = Transfer(targetAccount, amount); if (success) { Console.WriteLine($"📝 轉帳備註：{note}"); } return success; } // 查詢餘額方法 public decimal GetBalance() { return Balance; } // 查詢餘額方法多載 - 格式化輸出 public string GetBalance(string format) { return format.ToLower() switch { "currency" => $"${Balance:N2}", "number" => Balance.ToString("N2"), "simple" => Balance.ToString(), _ => $"${Balance:N2}" }; } // 顯示帳戶資訊 public void ShowInfo() { Console.WriteLine($"📋 帳戶資訊："); Console.WriteLine($" 帳號：{AccountNumber}"); Console.WriteLine($" 持有人：{OwnerName}"); Console.WriteLine($" 類型：{GetTypeName(Type)}"); Console.WriteLine($" 餘額：${Balance:N2}"); Console.WriteLine($" 開戶日期：{CreatedDate:yyyy-MM-dd}"); Console.WriteLine($" 狀態：{(IsActive ? "啟用" : "停用")}"); } // 覆寫 ToString 方法 public override string ToString() { return $"{AccountNumber} - {OwnerName}（{GetTypeName(Type)}，${Balance:N2}）"; } } // 銀行系統類別 public class Bank { private List accounts; public string BankName { get; set; } public Bank(string bankName) { BankName = bankName; accounts = new List(); Console.WriteLine($"🏛️ 建立銀行：{bankName}"); } // 開戶方法 public BankAccount OpenAccount(string ownerName, decimal initialBalance = 0, AccountType type = AccountType.Basic) { var account = new BankAccount(ownerName, initialBalance, type); accounts.Add(account); return account; } // 關戶方法 public bool CloseAccount(string accountNumber) { var account = FindAccount(accountNumber); if (account != null) { account.IsActive = false; Console.WriteLine($"🔒 關閉帳戶：{accountNumber}"); return true; } else { Console.WriteLine($"❌ 找不到帳戶：{accountNumber}"); return false; } } // 查詢帳戶 public BankAccount FindAccount(string accountNumber) { return accounts.Find(a => a.AccountNumber == accountNumber); } // 列出所有帳戶 public void ListAllAccounts() { if (accounts.Count == 0) { Console.WriteLine("🏦 銀行目前沒有帳戶"); return; } Console.WriteLine($"🏦 {BankName} 共有 {accounts.Count} 個帳戶："); for (int i = 0; i < accounts.Count; i++) { Console.WriteLine($"{i + 1}. {accounts[i]}"); } } // 統計資訊 public void ShowStatistics() { int totalAccounts = accounts.Count; int activeAccounts = accounts.Count(a => a.IsActive); decimal totalBalance = accounts.Sum(a => a.Balance); var typeStats = accounts.GroupBy(a => a.Type) .Select(g => new { Type = g.Key, Count = g.Count() }) .ToList(); Console.WriteLine($"📊 {BankName} 統計資訊："); Console.WriteLine($" 總帳戶數：{totalAccounts}"); Console.WriteLine($" 啟用帳戶：{activeAccounts}"); Console.WriteLine($" 總存款：${totalBalance:N2}"); Console.WriteLine(" 帳戶類型分布："); foreach (var stat in typeStats) { string typeName = stat.Type switch { AccountType.Basic => "基本帳戶", AccountType.Savings => "儲蓄帳戶", AccountType.Checking => "支票帳戶", _ => "未知" }; Console.WriteLine($" • {typeName}：{stat.Count}個"); } } } // 主程式 class Program { static void Main() { Console.WriteLine("=== 銀行系統示範 ==="); // 建立銀行 Bank bank = new Bank("台灣銀行"); // 開立不同類型的帳戶 BankAccount account1 = bank.OpenAccount("小明", 10000, AccountType.Savings); BankAccount account2 = bank.OpenAccount("小華", 5000, AccountType.Checking); BankAccount account3 = bank.OpenAccount("小美", 2000, AccountType.Basic); Console.WriteLine(); // 列出所有帳戶 bank.ListAllAccounts(); Console.WriteLine(); // 存款操作 account1.Deposit(5000); account1.Deposit(2000, "年終獎金"); account2.Deposit(3000); Console.WriteLine(); // 提款操作 account1.Withdraw(1000); account2.Withdraw(800, "繳房租"); account3.Withdraw(3000); // 餘額不足 Console.WriteLine(); // 轉帳操作 account1.Transfer(account3, 2000, "借錢給小美"); account2.Transfer(account1, 1000); Console.WriteLine(); // 查詢餘額 Console.WriteLine("查詢餘額："); Console.WriteLine($"小明：{account1.GetBalance("currency")}"); Console.WriteLine($"小華：{account2.GetBalance("currency")}"); Console.WriteLine($"小美：{account3.GetBalance("currency")}"); Console.WriteLine(); // 顯示帳戶詳細資訊 Console.WriteLine("帳戶詳細資訊："); account1.ShowInfo(); Console.WriteLine(); account2.ShowInfo(); Console.WriteLine(); account3.ShowInfo(); Console.WriteLine(); // 關閉一個帳戶 bank.CloseAccount(account2.AccountNumber); Console.WriteLine(); // 顯示統計資訊 bank.ShowStatistics(); } }
